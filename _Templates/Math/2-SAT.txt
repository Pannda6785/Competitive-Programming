/*
    n variables, SAT of disjunction clauses, ie. (-x or y) and (z or x)
    positive is [0, n)
    negative is [-n, 0)
*/
struct TwoSAT {
    int n;
    vector<vector<int>> adj;

    TwoSAT() : n(0) {}
    TwoSAT(int n) : n(n), adj(2 * n) {}

    int neg(int x) {
        if (x >= n) return x - n;
        return x + n;
    }

    void add(int x, int y) {
        x += n;
        y += n;
        adj[neg(x)].push_back(y);
        adj[neg(y)].push_back(x);
    }

    bool solve(vector<int> &solution) {
        vector<int> comp = [&]() -> vector<int> {
            vector<int> comp(2 * n, -1);
            int comp_idx = 0;
            vector<int> num(2 * n, -1), low(2 * n);
            int tim = 0;
            vector<int> stk;
            vector<bool> on_stk(2 * n, false);
            auto dfs = [&](auto self, int u) -> void {
                num[u] = low[u] = tim++;
                stk.push_back(u);
                on_stk[u] = true;
                for (int v : adj[u]) {
                    if (num[v] == -1) {
                        self(self, v);
                        low[u] = min(low[u], low[v]);
                    } else if (on_stk[v]) {
                        low[u] = min(low[u], num[v]);
                    }
                }
                if (num[u] == low[u]) {
                    int w;
                    do {
                        w = stk.back();
                        stk.pop_back();
                        on_stk[w] = false;
                        comp[w] = comp_idx;
                    } while (w != u);
                    comp_idx++;
                }
            };
            for (int u = 0; u < 2 * n; u++) {
                if (num[u] == -1) {
                    dfs(dfs, u);
                }
            }
            return comp;
        }();

        solution.resize(n);
        for (int u = 0; u < n; u++) {
            if (comp[u] == comp[neg(u)]) return false;
            solution[u] = comp[u] > comp[neg(u)] ? +1 : -1;
        }
        return true;
    }
};
